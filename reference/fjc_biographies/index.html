    def apply(self, sections: MutableMapping[str, str]) -> dict[str, Any]:
    """Applies every rule, in order, to `sections`.

    Args:
        sections: the sections of text, by name. Rules that make or change
            sections change it in place.

    Returns:
        The value of each variable, by name. A variable whose rules found
            nothing is False (flags), 0 (counts), an empty `list`
            (matches and names), or None (labels and sections).

    """
    results: dict[str, Any] = {}
    for rule in self.rules:
        rule.apply(sections, results)
    return results

    @classmethod
def load(cls, *sources: str | pathlib.Path) -> Rulebook:
    """Loads rules from CSV files.

    Args:
        *sources: names of built-in rulebooks (such as "judges"), or paths
            to CSV files or to folders of them. The rules are applied in
            the order they are given (and, in a folder, in the
            alphabetical order of the files). A file that two sources
            name is read only the first time.

    Returns:
        A rulebook with the rules of every source.

    Raises:
        FileNotFoundError: if a source is not a built-in rulebook or a file
            or folder.
        ValueError: if a file is missing a column or a row is not a valid
            rule. The message names the file and row.

    """
    rules: list[Rule] = []
    names: list[str] = []
    read: set[pathlib.Path] = set()
    for source in sources:
        for path in find_instructions(source):
            if path.resolve() in read:
                continue
            read.add(path.resolve())
            rules.extend(_read(path))
            names.append(path.stem)
    return cls(rules = rules, name = ', '.join(names))

    def add_jcs(
    roster: pd.DataFrame,
    folder: pathlib.Path | str | None = None,
    *,
    scores_folder: pathlib.Path | str | None = None,
    martin_quinn: pathlib.Path | str | None = None,
    update: bool = False) -> pd.DataFrame:
    """Adds each appointment's Judicial Common Space score as "jcs".

    The scores are computed by `judicial_common_space.compute` from the
    FJC's files in the "raw" folder of `folder` (the files that the roster
    was made from), Voteview's NOMINATE scores, the dates of senators'
    service, Wikipedia's lists of circuit judges' duty stations (if they
    can be downloaded), and the justices' Martin-Quinn scores (if they can
    be found).

    Args:
        roster: a roster with a row for each appointment, as
            `parser.build_roster` makes it.
        folder: the data folder with the FJC's files. Defaults to None, in
            which case `downloader.data_folder()` is used.
        scores_folder: the folder for the other files. Defaults to None, in
            which case the data folder of judicial-common-space is used.
        martin_quinn: the path of the file of the justices' Martin-Quinn
            scores. Defaults to None, in which case the file is downloaded
            if it is not there. Without it, the justices have no score.
        update: whether to download the files again even if they are there.
            Defaults to False.

    Returns:
        A new roster with "jcs" after "senate_vote". It is missing for an
            appointment that has no score (as for one by Herbert Hoover,
            whom Voteview does not score).

    Raises:
        KeyError: if `roster` lacks "nid" or "sequence".

    """
    _require(roster, ['nid', 'sequence'])
    files = jcs.sources.paths(scores_folder)
    files.update(_fjc_files(folder, update = update))
    jcs.sources.download(_POLITICS, scores_folder, overwrite = update)
    try:
        jcs.sources.download(
            'duty_stations', scores_folder, overwrite = update)
    except Exception as error:  # noqa: BLE001
        logger.warning(
            'could not download the duty stations of circuit judges, so '
            'judges whose careers show no state have their birth states: %s',
            error)
    try:
        found = _martin_quinn_file(martin_quinn, scores_folder, update = update)
    except RuntimeError as error:
        logger.warning('%s, so the justices have no scores', error)
        found = None
    scored, _ = jcs.compute(files, martin_quinn = found)
    keyed = scored[['nid', 'sequence', 'jcs']].assign(
        nid = pd.to_numeric(scored['nid']).astype('Int64'),
        sequence = pd.to_numeric(scored['sequence']).astype('Int64'),
        jcs = pd.to_numeric(scored['jcs']).astype('Float64'))
    return _insert(roster, keyed, ['jcs'])

            folder, scores_folder = scores_folder,
            martin_quinn = martin_quinn, update = update)

    build_roster(service, demographics, career=None, *, rulebook='judges') ¶

    Makes a table with a row for each court that each judge has served on.

    The table has these columns:

    Column Meaning
    nid The FJC's number for the judge.
    judge The judge's name, as the FJC reports it.
    last_name, first_name, middle_name, suffix Its parts.
    name_first_middle_last, ... The name written in each of the ways that names.permutations writes it, for matching it against other sources: with and without the middle name or its initial and the suffix, and with the letters that the FJC puts in square brackets kept or replaced with a period (see names).
    sequence The FJC's number for the order of the appointment among the judge's, which identifies it with nid.
    court, court_type The court's name and its type.
    circuit_num The number of the court's circuit: its own for a court of appeals (12 for the D.C. Circuit and 13 for the Federal Circuit), or for a district court, that of its state now. Missing for other courts.
    start_date, end_date, senior_date The dates the judge's service on the court began (with a commission or a recess appointment) and ended, and the judge took senior status. end_date is missing for a judge who is still serving.
    start_year, end_year, senior_year Their years.
    birth_year The year the judge was born.
    president The president who appointed the judge.
    party That president's party: -1 for Democratic and 1 for Republican. Missing for other parties.
    aba_rating The American Bar Association's rating, from 1 (not qualified) to 4 (exceptionally well qualified). Missing if there was none, or it was split ("Qualified/Well Qualified").
    recess Whether the judge first took the seat by a recess appointment.
    senate_vote The share of senators who voted to confirm the judge (1 for a voice vote). Missing if the vote is not recorded.
    woman, minority Whether the judge is a woman, and whether the judge's race or ethnicity is anything other than white. Missing if the FJC does not report it.

    A judge who was reassigned to another court without a new appointment (as when the Eleventh Circuit was divided from the Fifth) keeps the president, party, rating, recess appointment, and vote of the judge's last appointment.

    If career is passed, the judge's professional career is the column "career", with each position on its own line, in order. It is the text that the rules search, so the fields made from it can be checked against it. Each variable that rulebook makes from the career (see code_careers) is a column after it. The "judges" rulebook makes "prosecutor", "public_defender", "law_clerk", "supreme_court_clerk", "solicitor_general", and "law_professor". "career" is missing for a judge with no career in the file.

    def build_roster(
    service: pd.DataFrame | pathlib.Path | str,
    demographics: pd.DataFrame | pathlib.Path | str,
    career: pd.DataFrame | pathlib.Path | str | None = None,
    *,
    rulebook: rules.Rulebook | pathlib.Path | str = 'judges') -> pd.DataFrame:
    """Makes a table with a row for each court that each judge has served on.

    The table has these columns:

    | Column | Meaning |
    | --- | --- |
    | `nid` | The FJC's number for the judge. |
    | `judge` | The judge's name, as the FJC reports it. |
    | `last_name`, `first_name`, `middle_name`, `suffix` | Its parts. |
    | `name_first_middle_last`, ... | The name written in each of the ways that `names.permutations` writes it, for matching it against other sources: with and without the middle name or its initial and the suffix, and with the letters that the FJC puts in square brackets kept or replaced with a period (see `names`). |
    | `sequence` | The FJC's number for the order of the appointment among the judge's, which identifies it with `nid`. |
    | `court`, `court_type` | The court's name and its type. |
    | `circuit_num` | The number of the court's circuit: its own for a court of appeals (12 for the D.C. Circuit and 13 for the Federal Circuit), or for a district court, that of its state now. Missing for other courts. |
    | `start_date`, `end_date`, `senior_date` | The dates the judge's service on the court began (with a commission or a recess appointment) and ended, and the judge took senior status. `end_date` is missing for a judge who is still serving. |
    | `start_year`, `end_year`, `senior_year` | Their years. |
    | `birth_year` | The year the judge was born. |
    | `president` | The president who appointed the judge. |
    | `party` | That president's party: -1 for Democratic and 1 for Republican. Missing for other parties. |
    | `aba_rating` | The American Bar Association's rating, from 1 (not qualified) to 4 (exceptionally well qualified). Missing if there was none, or it was split ("Qualified/Well Qualified"). |
    | `recess` | Whether the judge first took the seat by a recess appointment. |
    | `senate_vote` | The share of senators who voted to confirm the judge (1 for a voice vote). Missing if the vote is not recorded. |
    | `woman`, `minority` | Whether the judge is a woman, and whether the judge's race or ethnicity is anything other than white. Missing if the FJC does not report it. |

    A judge who was reassigned to another court without a new appointment
    (as when the Eleventh Circuit was divided from the Fifth) keeps the
    president, party, rating, recess appointment, and vote of the judge's
    last appointment.

    If `career` is passed, the judge's professional career is the column
    "career", with each position on its own line, in order. It is the text
    that the rules search, so the fields made from it can be checked
    against it. Each variable that `rulebook` makes from the career (see
    `code_careers`) is a column after it. The "judges" rulebook makes
    "prosecutor", "public_defender", "law_clerk", "supreme_court_clerk",
    "solicitor_general", and "law_professor". "career" is missing for a
    judge with no career in the file.

    Args:
        service: the file of federal judicial service, as a path, the table
            that was read from it, or its tidy table.
        demographics: the file of demographics, in any of those forms.
        career: the file of professional careers, in any of those forms.
            Defaults to None, in which case no careers are coded.
        rulebook: the rules that code the careers: a rulebook, or the name of
            a built-in rulebook or the path of a CSV file. Defaults to
            "judges".

    Returns:
        The table, sorted by judge and then by the order of the judge's
            service.

    Raises:
        KeyError: if a file lacks a column that is needed.
        ValueError: if the file of demographics has more than one row for a
            judge, or a variable of `rulebook` has the name of a column of
            the table.

    """
    served = _table(service)
    _require(served, _SERVICE, 'the file of federal judicial service')
    people = _table(demographics)
    _require(people, _PERSONAL, 'the file of demographics')
    order = ['nid', 'sequence'] if 'sequence' in served.columns else ['nid']
    served = served.sort_values(order, kind = 'stable').reset_index(drop = True)
    # A reassigned judge has "None (reassignment)" in place of a president,
    # so the columns of the appointment are taken from the judge's last row
    # that is not a reassignment.
    reassigned = (
        served['appointing_president'].fillna('').str.startswith('None')
        .to_numpy())
    own = pd.Series(np.arange(len(served)), dtype = 'float64').mask(reassigned)
    source = (
        own.groupby(served['nid'], dropna = False).ffill()
        .fillna(-1).astype(int))
    columns = list(_APPOINTMENT)
    appointed = served[columns].reindex(source.to_numpy())
    appointed.index = served.index
    served[columns] = appointed
    # Each row of a judge's service gets the judge's demographics. A judge
    # who is not in the file of demographics has missing ones.
    personal = served[['nid']].merge(
        people[list(_PERSONAL)], on = 'nid', how = 'left',
        validate = 'many_to_one')
    started = served['commission_date'].fillna(
        served['recess_appointment_date'])
    gender = personal['gender'].map({'Female': True, 'Male': False})
    race = personal['race_or_ethnicity']
    roster = pd.DataFrame({
        'nid': served['nid'],
        'judge': served['judge_name'],

    def roster_by_judge(roster: pd.DataFrame) -> pd.DataFrame:
    """Returns a roster with one row for each judge.

    A roster made by `build_roster` has a row for each appointment: each
    court that each judge has served on. This is the same information
    organized by judge, as the FJC's own "organized by judge" export is. The
    columns that describe the judge (such as the judge's name, "birth_year",
    "woman", and the fields of the judge's career) are given once.
    "appointments" is how many courts the judge has served on. Then the
    columns that describe an appointment (such as "court", "start_date", and
    "party") are given for each, numbered in the order of the judge's
    service: "court_1", "start_date_1", ..., "court_2", and so on. They are
    missing for a judge with fewer appointments than the most that any
    judge has.

    Args:
        roster: a roster with a row for each appointment, as `build_roster`
            makes it, in the order of each judge's service.

    Returns:
        The roster with a row for each judge, in the order that the judges
            first appear in `roster`.

    Raises:
        KeyError: if `roster` lacks "nid" or "court".

    """
    _require(roster, ['nid', 'court'], 'a roster')
    roster = roster.dropna(subset = ['nid']).reset_index(drop = True)
    fields = [c for c in _BY_APPOINTMENT if c in roster.columns]
    personal = [c for c in roster.columns if c not in _BY_APPOINTMENT]
    judges = roster[personal].drop_duplicates('nid').set_index('nid')
    judges.insert(
        len(judges.columns), 'appointments',
        roster.groupby('nid').size().reindex(judges.index).astype('int64'))
    place = roster.groupby('nid').cumcount() + 1
    # A boolean column gets missing values for the judges with fewer
    # appointments, so it is made one that can have them.
    appointed = roster[fields].astype({
        c: 'boolean' for c in fields if roster[c].dtype == bool})
    # `pivot_table` would aggregate each cell, which loses text and types, so
    # each judge's appointments are unstacked instead.
    indexed = appointed.set_index([roster['nid'], place.rename('place')])
    wide = pd.DataFrame(indexed.unstack('place'))  # noqa: PD010
    numbered = [
        (field, number) for number in range(1, int(place.max() or 0) + 1)
        for field in fields]
    wide = wide.reindex(columns = pd.MultiIndex.from_tuples(numbered))
    wide.columns = [f'{field}_{number}' for field, number in numbered]
    return pd.concat(
        [judges, wide.reindex(judges.index)], axis = 1).reset_index()

    def run(
    folder: pathlib.Path | str | None = None,
    *,
    update: bool = False,
    rulebook: rules.Rulebook | pathlib.Path | str = 'judges',
    jcs: bool = False,
    martin_quinn: bool | pathlib.Path | str = False,
    scores_folder: pathlib.Path | str | None = None) -> dict[
        str, pathlib.Path]:
    """Downloads, parses, and exports the FJC's files.

    Args:
        folder: the data folder. Defaults to None, in which case
            `downloader.data_folder()` is used.
        update: whether to download the files again even if they are there,
            to get the judges and corrections since they were downloaded.
            Defaults to False.
        rulebook: the rules that make the roster's fields about judges'
            careers (see `parser.code_careers`). Defaults to "judges".
        jcs: whether to add each appointment's Judicial Common Space score
            (see `scores.add_jcs`). Defaults to False.
        martin_quinn: whether to add the justices' Martin-Quinn scores (see
            `scores.add_martin_quinn`) and export their scores in each term,
            or the path of the file of the scores ("justices.csv"), which is
            also used for the justices' Judicial Common Space scores.
            Defaults to False.
        scores_folder: the folder for the files that the scores come from.
            Defaults to None, in which case the data folder of
            judicial-common-space is used.

    Returns:
        The path of each CSV file that was saved, by its category, the
            paths of the roster as "roster_by_appointment" and
            "roster_by_judge", and with `martin_quinn`, the path of the
            justices' scores in each term as "martin_quinn_terms".

    Examples:
        >>> import fjc_biographies
        >>> fjc_biographies.run(update = True)  # doctest: +SKIP

    """
    root = pathlib.Path(folder).expanduser() if folder else (
        downloader.data_folder())
    tables = parser.load_all(root, update = update)
    roster = parser.build_roster(
        tables['service'], tables['demographics'], tables['career'],
        rulebook = rulebook)
    # A path names the file of Martin-Quinn scores, and True finds it.
    path = None if isinstance(martin_quinn, bool) else martin_quinn
    if jcs:
        roster = scores.add_jcs(
            roster, root, scores_folder = scores_folder, martin_quinn = path,
            update = update)
    terms = None
    if martin_quinn:
        terms = scores.martin_quinn_terms(
            root, scores_folder = scores_folder, martin_quinn = path,
            update = update and not jcs)
        roster = scores.add_martin_quinn(roster, terms = terms)
    tables['roster_by_appointment'] = roster
    tables['roster_by_judge'] = parser.roster_by_judge(roster)
    if terms is not None:
        tables['martin_quinn_terms'] = terms
    return {
        name: export(table, root / f'{name}.csv')
        for name, table in tables.items()}

    def tidy(table: pd.DataFrame) -> pd.DataFrame:
    """Returns a table of the directory that is easy to use.

    A tidy table is not changed, so a table can be tidied more than once.

    Args:
        table: a table of one of the FJC's files, as it was read.

    Returns:
        A new table whose columns have lower case names, with underscores in
            place of spaces and punctuation (and some shorter names, such as
            "position" for "Professional Career", which are in
            `options._COLUMN_NAMES`). Text has no spaces at its ends, and a
            blank cell is missing. "nid", "jid", "sequence", and the months
            and days of births and deaths are whole numbers, as are the
            years, and the columns whose names end in "_date" are dates.
            The empty, unnamed columns at the end of the file of other
            federal judicial service are left out.

    """
    tidied: dict[str, pd.Series] = {}
    for column in table.columns:
        name = _column_name(column)
        values = table[column]
        if name in _NUMBERS:
            values = _number(values, r'(\d+)')
        elif name in _YEARS:
            values = _number(values, r'(\d{4})')
        elif pd.api.types.is_datetime64_any_dtype(values.dtype):
            pass
        elif name.endswith('_date'):
            values = pd.to_datetime(
                _text(values), format = 'mixed', errors = 'coerce')
        elif not pd.api.types.is_numeric_dtype(values.dtype):
            values = _text(values)
        unnamed = not name or name.startswith('unnamed_')
        if unnamed and values.isna().all():
            continue
        tidied[name] = values
    return pd.DataFrame(tidied, index = table.index)
